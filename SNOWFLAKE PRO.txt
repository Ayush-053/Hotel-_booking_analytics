CREATE DATABASE HOTEL_DB;

CREATE OR REPLACE FILE FORMAT HOTEL_CSV
    TYPE = 'CSV'
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    SKIP_HEADER = 1
    NULL_IF = ('NULL','null','')

CREATE OR REPLACE STORAGE INTEGRATION HOTEL_S3_INTEGRATION
TYPE = EXTERNAL_STAGE
STORAGE_PROVIDER = S3
ENABLED = TRUE
STORAGE_AWS_ROLE_ARN = 'arn:aws:iam::232025405371:role/hotel_pvt'
STORAGE_ALLOWED_LOCATIONS = (
    's3://hotel-booking-data1/hotel_bookings_raw.csv'
);

DESC STORAGE INTEGRATION HOTEL_S3_INTEGRATION;

CREATE STAGE HOTEL_EX
    URL = 's3://hotel-booking-data1/hotel_bookings_raw.csv'
    STORAGE_INTEGRATION = HOTEL_S3_INTEGRATION
    FILE_FORMAT = (FILE_FORMAT = HOTEL_CSV) ;


LIST @HOTEL_DB.PUBLIC.HOTEL_EX;

SELECT $1,$2,$3,$4,$5,$6,$7,$8,$10,$11,$12,$13,$14 FROM @HOTEL_DB.PUBLIC.HOTEL_EX;


CREATE TABLE HOTEL_DB.PUBLIC.BRONZE_TBL(
    booking_id STRING,
    hotel_id STRING,
    hotel_city STRING,
    customer_id	STRING,
    customer_name STRING,
    customer_email STRING,
    check_in_date STRING,
    check_out_date STRING,
    room_type STRING,
    num_guests STRING,
    total_amount STRING,
    currency STRING,
    booking_status STRING
);

CREATE OR REPLACE PIPE HOTEL_DB.PUBLIC.HOTEL_PIPE
AUTO_INGEST = TRUE
AS
COPY INTO HOTEL_DB.PUBLIC.BRONZE_TBL
FROM @HOTEL_EX
FILE_FORMAT = (FORMAT_NAME = HOTEL_CSV)
ON_ERROR = 'CONTINUE';

DESC PIPE HOTEL_DB.PUBLIC.HOTEL_PIPE;

SHOW PIPES LIKE 'HOTEL_PIPE';

SELECT SYSTEM$PIPE_STATUS(
    'HOTEL_DB.PUBLIC.HOTEL_PIPE'
);

SELECT * FROM HOTEL_DB.PUBLIC.BRONZE_TBL LIMIT 10;

CREATE TABLE HOTEL_DB.PUBLIC.SILVER_TBL(
    booking_id VARCHAR,
    hotel_id VARCHAR,
    hotel_city VARCHAR,
    customer_id	VARCHAR,
    customer_name VARCHAR,
    customer_email VARCHAR,
    check_in_date DATE,
    check_out_date DATE,
    room_type VARCHAR,
    num_guests VARCHAR,
    total_amount FLOAT,
    currency VARCHAR,
    booking_status VARCHAR
);

SELECT CUSTOMER_EMAIL
FROM HOTEL_DB.PUBLIC.BRONZE_TBL
WHERE NOT (CUSTOMER_EMAIL LIKE '%@%.%')
    OR CUSTOMER_EMAIL IS NULL;

SELECT TOTAL_AMOUNT FROM HOTEL_DB.PUBLIC.BRONZE_TBL
    WHERE TRY_TO_NUMBER(TOTAL_AMOUNT) < 0;

SELECT CHECK_IN_DATE,CHECK_OUT_DATE FROM HOTEL_DB.PUBLIC.BRONZE_TBL
    WHERE TRY_TO_DATE(CHECK_OUT_DATE) < TRY_TO_DATE(CHECK_IN_DATE);

SELECT DISTINCT(BOOKING_STATUS) FROM HOTEL_DB.PUBLIC.BRONZE_TBL;

INSERT INTO HOTEL_DB.PUBLIC.SILVER_TBL
SELECT
    booking_id ,
    hotel_id,
    INITCAP(TRIM(hotel_city)) AS hotel_city,
    customer_id,
    INITCAP(TRIM(customer_name)),
    CASE 
        WHEN customer_email LIKE '%@%.%' THEN LOWER(TRIM(customer_email))
    ELSE NULL
    END AS customer_email,
    TRY_TO_DATE(NULLIF(check_in_date, '')) AS check_in_date,
    TRY_TO_DATE(NULLIF(check_out_date, '')) AS check_out_date,
    room_type,
    num_guests,
    ABS(TRY_TO_NUMBER(total_amount)) AS total_amount,
    currency,
    CASE
        WHEN LOWER(booking_status) IN ('Confirmeeed', 'Confirmd') THEN 'Confirmed'
        ELSE booking_status
    END AS booking_status
    FROM BRONZE_TBL
    WHERE
        TRY_TO_DATE(check_in_date) IS NOT NULL
        AND TRY_TO_DATE(check_out_date) IS NOT NULL 
        AND TRY_TO_DATE(check_out_date) >= TRY_TO_DATE(check_in_date);

SELECT * FROM HOTEL_DB.PUBLIC.SILVER_TBL LIMIT 50;

CREATE TABLE GOLDEN_DAILY_BOOKING AS
SELECT
    check_in_date AS DATE,
    COUNT(*) AS Total_Booking,
    SUM(TOTAL_AMOUNT) AS Total_Revenue
FROM HOTEL_DB.PUBLIC.SILVER_TBL
GROUP BY check_in_date
ORDER BY DATE;

CREATE TABLE GOLDEN_HOTEL_CITY_SALES AS 
SELECT 
    hotel_city,
    SUM(TOTAL_AMOUNT) AS Total_Revenue
FROM HOTEL_DB.PUBLIC.SILVER_TBL
GROUP BY hotel_city
ORDER BY Total_Revenue DESC;

CREATE TABLE GOLDEN_CLEAN_BOOKING AS 
SELECT 
    booking_id,
    hotel_id,
    hotel_city,
    customer_id,
    customer_name,
    customer_email,
    check_in_date,
    check_out_date,
    room_type,
    num_guests,
    total_amount,
    currency,
    booking_status
FROM HOTEL_DB.PUBLIC.SILVER_TBL;

SELECT * FROM GOLDEN_DAILY_BOOKING LIMIT 30;

SELECT * FROM GOLDEN_HOTEL_CITY_SALES LIMIT 30;
